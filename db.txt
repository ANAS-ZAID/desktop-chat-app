--استرجاع قائمة المحادثات حسب المستخدم الحالي
CREATE PROCEDURE GetChatList
    @CurrentUserPhone INT
AS
BEGIN
    WITH LatestMessages AS (
        SELECT 
            m.id,
            m.containts AS MessageContent,
            m.status AS Status,
            m.startDate AS SentDate,
            m.reciveDate AS ReceivedDate,
            m.sendPhone AS SenderPhone,
            m.recivePhone AS ReceiverPhone,
            m.reciveGroupId AS ReceiverGroupId,
            u1.name AS SenderName,
            u2.name AS ReceiverName,
			case 
			when m.reciveGroupId is not NULL then m.reciveGroupId
			else
			m.recivePhone
			end as  ChatId,
            g.name AS GroupName,
            g.img AS GroupImage,
            u1.img AS SenderImage,
			 u2.active AS ReceiverActive,
			 u2.lastSeen AS lastSeen,
            ROW_NUMBER() OVER (
                PARTITION BY 
                    CASE 
                        WHEN m.reciveGroupId IS NOT NULL THEN m.reciveGroupId 
                        ELSE LEAST(m.sendPhone, m.recivePhone) 
                    END,
                    CASE 
                        WHEN m.reciveGroupId IS NOT NULL THEN m.reciveGroupId 
                        ELSE GREATEST(m.sendPhone, m.recivePhone) 
                    END
                ORDER BY m.startDate DESC
            ) AS RowNum
        FROM messages m
        LEFT JOIN users u1 ON m.sendPhone = u1.phone
        LEFT JOIN users u2 ON m.recivePhone = u2.phone

        LEFT JOIN groups g ON m.reciveGroupId = g.id
        WHERE 
            m.sendPhone = @CurrentUserPhone 
            OR m.recivePhone = @CurrentUserPhone 
            OR m.reciveGroupId IN (SELECT groupId FROM memeparsGroups WHERE userPhone = @CurrentUserPhone)
    )
    SELECT 
        id, MessageContent, SentDate, SenderName, ReceiverName,ChatId, GroupName,

        CASE 
            WHEN ReceiverGroupId IS NOT NULL THEN GroupImage
            ELSE SenderImage
        END AS ChatImage,
        CASE 
            WHEN ReceiverGroupId IS NOT NULL THEN 'Group'
            ELSE 'Private'
        END AS MessageType
    FROM LatestMessages
    WHERE RowNum = 1
    ORDER BY SentDate DESC;
END;
execute GetChatList @CurrentUserPhone=730000001;

CREATE PROCEDURE GetMessagesByChatId
    @CurrentUserPhone INT,
    @ChatId INT
AS
BEGIN
    SELECT 
	    m.id,
        m.containts AS MessageContent,
        m.startDate AS SentDate,
		m.reciveDate AS ReceivedDate,
        u.elesName AS SenderName,
        u.img AS SenderImage,
        m.status AS Status,
		m.sendPhone,
		case
		when m.reciveGroupId IS NOT NULL then m.reciveGroupId
		else m.recivePhone 
		end as receiverId,

        CASE 
            WHEN ma.attachmentType IS NOT NULL THEN ma.filePath
            ELSE NULL
        END AS Attachment
    FROM messages m
    LEFT JOIN users u ON m.sendPhone = u.phone
    LEFT JOIN MessageAttachments ma ON m.id = ma.messageId
    WHERE 
        (  @CurrentUserPhone IN (m.sendPhone, m.recivePhone)	And m.recivePhone = @ChatId  )
        OR (m.reciveGroupId = @ChatId and  m.reciveGroupId IN (SELECT groupId FROM memeparsGroups WHERE userPhone = @CurrentUserPhone))
    ORDER BY m.startDate ASC;
END;

CREATE PROCEDURE SendMessage
    @CurrentUserPhone INT,
    @MessageContent NVARCHAR(MAX),
    @ReceiverPhone INT = NULL,
    @ReceiverGroupId INT = NULL
AS
BEGIN
    IF NOT EXISTS (SELECT 1 FROM users WHERE phone = @CurrentUserPhone)
    BEGIN
        RAISERROR ('المستخدم غير موجود', 16, 1);
        RETURN;
    END

    IF @ReceiverPhone IS NOT NULL AND NOT EXISTS (SELECT 1 FROM users WHERE phone = @ReceiverPhone)
    BEGIN
        RAISERROR ('المستلم غير موجود', 16, 1);
        RETURN;
    END

    IF @ReceiverGroupId IS NOT NULL AND NOT EXISTS (SELECT 1 FROM groups WHERE id = @ReceiverGroupId)
    BEGIN
        RAISERROR ('المجموعة غير موجودة', 16, 1);
        RETURN;
    END

    INSERT INTO messages (containts, startDate, sendPhone, recivePhone, reciveGroupId)
    VALUES (@MessageContent, GETDATE(), @CurrentUserPhone, @ReceiverPhone, @ReceiverGroupId);
END;

CREATE PROCEDURE SetUserStatus
    @UserPhone INT,
    @Status NVARCHAR(20)
AS
BEGIN
    IF EXISTS (SELECT 1 FROM UserStatus WHERE userPhone = @UserPhone)
    BEGIN
        UPDATE UserStatus
        SET status = @Status, lastUpdate = GETDATE()
        WHERE userPhone = @UserPhone;
    END
    ELSE
    BEGIN
        INSERT INTO UserStatus (userPhone, status)
        VALUES (@UserPhone, @Status);
    END
END;

CREATE PROCEDURE MarkMessageAsRead
    @MessageId INT,
    @UserPhone INT
AS
BEGIN
    IF NOT EXISTS (SELECT 1 FROM MessageReadStatus WHERE messageId = @MessageId AND userPhone = @UserPhone)
    BEGIN
        INSERT INTO MessageReadStatus (messageId, userPhone, readStatus, readDate)
        VALUES (@MessageId, @UserPhone, 1, GETDATE());
    END
    ELSE
    BEGIN
        UPDATE MessageReadStatus
        SET readStatus = 1, readDate = GETDATE()
        WHERE messageId = @MessageId AND userPhone = @UserPhone;
    END
END;
